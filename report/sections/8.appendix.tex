\section{Supplementary derivations and theoretical extensions}
\label{sec:appendix}

This appendix collects derivations and complementary results that support the main development
without altering the inferential scope.
The emphasis is on (i) making key steps in the conjugate-exponential-family integration fully
explicit, (ii) clarifying the probabilistic structure of factorizing partition priors, and (iii)
recording a few boundary cases and identifiability remarks that are useful in practice.

\subsection{Conjugate exponential families: detailed block evidence derivation}
\label{app:conjugate}

This appendix gives a step-by-step proof of Theorem~\ref{thm:ef-integral} (closed-form
EF--conjugate block evidence) and the companion posterior-moment formulas of
Section~\ref{sec:ef}.

\begin{proof}[Proof of Theorem~\ref{thm:ef-integral}]
\ProofStep{Setup.}
Fix a segment (block) $(i,j]$ and write its aggregated sufficient statistics
$S_{ij}=\sum_{t=i+1}^j w_t T(y_t)$, aggregated exposure
$W_{ij}=\sum_{t=i+1}^j w_t$, and log base-measure sum
$H_{ij}=\sum_{t=i+1}^j \log h(y_t,w_t)$.
Let $\eta\in\mathbb{R}^d$ denote the natural parameter and $A(\eta)$ its cumulant function.
The conditional likelihood factors as
\begin{equation}
 p\big(y_{(i,j]}\mid \eta\big)
 \;=\; \prod_{t=i+1}^j h(y_t,w_t)\exp\big\{ w_t\,\langle \eta, T(y_t)\rangle - w_t A(\eta)\big\}
 \;=\; \exp\big\{H_{ij}+\langle \eta, S_{ij}\rangle - W_{ij}A(\eta)\big\}.
 \label{eq:app-ef-lik}
\end{equation}
This uses the same weighted base-measure convention as the statement of Theorem~\ref{thm:ef-integral}.
The standard Diaconis--Ylvisaker conjugate prior takes the form
\begin{equation}
 p(\eta\mid \tau,\nu) \;=\; \exp\big\{\langle \eta,\tau\rangle - \nu A(\eta) - \log Z(\tau,\nu)\big\},
 \label{eq:app-ef-prior}
\end{equation}
where $(\tau,\nu)$ are hyperparameters and $Z(\tau,\nu)$ is the normalizer. This appendix uses
$(\tau,\nu)$ for the same conjugate-prior quantities denoted $(\alpha,\beta)$ in the main text:
$\tau=\alpha$ and $\nu=\beta$ before observing the block, and the posterior update is
$(\tau+S_{ij},\nu+W_{ij})$.

\ProofStep{Evidence integral.}
Multiplying \eqref{eq:app-ef-lik} and \eqref{eq:app-ef-prior} yields
\begin{align}
 p\big(y_{(i,j]}\mid \eta\big)\,p(\eta\mid\tau,\nu)
 &= \frac{\exp\{H_{ij}\}}{Z(\tau,\nu)}
 \exp\Big\{\langle \eta, \tau+S_{ij}\rangle - (\nu+W_{ij})A(\eta)\Big\}.
\end{align}
Integrating over $\eta$ therefore gives
\begin{align}
 A^{(0)}_{ij}
 &= \int p\big(y_{(i,j]}\mid \eta\big)\,p(\eta\mid\tau,\nu)\,d\eta \\
 &= \exp\{H_{ij}\}\,\frac{Z\big(\tau+S_{ij},\nu+W_{ij}\big)}{Z(\tau,\nu)}.
 \label{eq:app-evidence}
\end{align}
This is the closed form claimed by Theorem~\ref{thm:ef-integral}, written at the level of a fixed block, and completes the proof. \end{proof}

\paragraph{Posterior distribution and moments.}
Bayes' rule shows that the posterior remains in the conjugate family:
\begin{equation}
 p(\eta\mid y_{(i,j]},\tau,\nu) \;\propto\; \exp\big\{\langle \eta,\tau+S_{ij}\rangle - (\nu+W_{ij})A(\eta)\big\}
 \;=\; p\big(\eta\mid \tau+S_{ij},\nu+W_{ij}\big).
 \label{eq:app-post}
\end{equation}
When differentiability and integrability permit interchange of differentiation and integration,
posterior moments of the relevant expectation parameter can be obtained by differentiating
$\log Z(\tau,\nu)$. In particular, $\nabla_{\tau}\log Z(\tau,\nu)$ yields the conjugate-family
mean of that expectation parameter, and higher derivatives yield higher cumulants. For families
where the Bayes-curve target is not this expectation parameter, the family-specific formulas in
Section~\ref{sec:families} provide the required formulas. The block-specific moment numerators
$A^{(1)}_{ij},A^{(2)}_{ij}$ are obtained by multiplying the appropriate posterior moments by the
evidence $A^{(0)}_{ij}$.

\subsection{Factorizing partition priors as renewal processes}
\label{app:renewal}

Assumption~\ref{ass:factorizing-prior} states that, conditional on the segment count $k$,
the boundary vector $t=(t_0,\dots,t_k)$ has a density proportional to a product of segment-length factors.
A translation-invariant subclass coincides with renewal-process models on the discrete index set;
more general irregular-design cohesions are product-partition priors rather than
iid renewal laws.

\begin{definition}[Renewal-process segmentation prior]
\label{def:renewal}
Fix $k\ge1$, and assume the design is translation-invariant in the sense that
$g(\Delta_x(i,i+\ell))$ depends on $i$ only through $\ell$ (e.g., equispaced boundary coordinates with
$u_i=i$, so $g(\ell)$ is a function of the integer segment length alone). Let $L_1,\dots,L_k$ be
positive integer-valued segment lengths with unnormalized mass function $\varphi(\ell):=g(\ell)$.
A \emph{renewal segmentation} draws lengths $(L_q)_{q=1}^k$ and sets boundaries
$t_q := \sum_{r=1}^q L_r$ with the constraint $t_k=n$.
\end{definition}

\begin{proposition}[Equivalence of renewal and factorized boundary priors]
\label{prop:renewal-equivalence}
Under Definition~\ref{def:renewal} and the translation-invariance hypothesis,
the induced distribution on $t=(t_0,\dots,t_k)$ satisfies
$p(t\mid k)\propto \prod_{q=1}^k g(\Delta_x(t_{q-1},t_q))$, which reduces to the displayed integer-length form on an equispaced grid. Conversely, any prior of the form
\eqref{eq:lengthprior} \emph{whose length factor depends only on the segment length} can be
realized as a renewal segmentation prior with $\varphi(\ell)=g(\ell)$.
\end{proposition}

\begin{proof}
For a fixed boundary vector $t$, the corresponding length sequence is uniquely determined as
$L_q=t_q-t_{q-1}$. Under translation invariance the renewal construction assigns that sequence
mass proportional to $\prod_{q=1}^k \varphi(L_q)=\prod_{q=1}^k g(L_q)$. This matches
$p(t\mid k)\propto\prod_{q=1}^k g(\Delta_x(t_{q-1},t_q))$, proving the forward direction under the equispaced translation-invariant convention. The converse
sets $\varphi(\ell):=g(\ell)$ as the unnormalized length mass.
\end{proof}

\begin{remark}[Hidden-Markov interpretation]
For a true renewal segmentation, the process can be represented as an HMM whose latent state tracks
the ``age'' within the current segment (or, equivalently, the last boundary location). Under those
renewal conditions, the forward recursion in Section~\ref{sec:dp} coincides with the standard
forward algorithm after integrating out segment parameters into block evidences. For arbitrary
irregular-design product cohesions this provides intuition rather than an exact HMM equivalence.
\end{remark}

\subsection{Posterior boundary and segment-membership marginals}
\label{app:marginals}

Theorem~\ref{thm:dp-correctness} gives closed forms for boundary marginals and for the segment-membership probability
$p\big(t\in(i,j],\,\text{segment }(i,j]\mid y,k\big)$.
We spell out the algebra that leads to these expressions.

\begin{proposition}[Boundary marginals from prefix/suffix evidences]
\label{prop:boundary-marginals}
Fix $k\ge1$ and suppose Assumption~\ref{ass:factorizing-prior} holds.
Let $\widetilde{L}_{r,j}$ and $\widetilde{R}_{r,j}$ be the prefix and suffix evidences defined in \eqref{eq:LR}.
Then for any $t\in\{1,\dots,n-1\}$,
\begin{equation}
 p(t\text{ is a boundary}\mid y,k)
 \;=\; \sum_{r=1}^{k-1}\ \frac{\widetilde{L}_{r,t}\,\widetilde{R}_{k-r,t}}{\widetilde{L}_{k,n}}.
 \label{eq:app-boundary-marginal}
\end{equation}
\end{proposition}

\begin{proof}
By definition, $t$ is a boundary at position $r$ iff $t_r=t$.
For fixed $r$, summing the joint posterior over all boundary vectors with $t_r=t$ yields
\begin{align}
 p(t_r=t\mid y,k)
 &\propto \sum_{\substack{0<t_1<\cdots<t_{k-1}<n\\ t_r=t}}
 \prod_{q=1}^k \widetilde{A}^{(0)}_{t_{q-1},t_q}.
\end{align}
The constraint $t_r=t$ splits the product into a prefix part (segments $1{:}r$ ending at $t$) and a suffix part (segments $r+1{:}k$ starting at $t$).
Summing over all admissible prefixes gives $\widetilde{L}_{r,t}$ by definition of the DP; summing over all admissible suffixes gives $\widetilde{R}_{k-r,t}$.
Note that the partition-prior normalizer $C_k$ multiplies the numerator for every boundary vector
under consideration (because $p(t\mid k)\propto C_k^{-1}\prod_q g(\cdot)$ for each $t$), and
multiplies the denominator through the same factor on the total evidence $p(y\mid k)$; it
therefore cancels, and \eqref{eq:app-boundary-marginal} involves only quantities that can be
computed from $\widetilde L,\widetilde R$.
Normalizing by the total evidence $\widetilde{L}_{k,n}$ yields \eqref{eq:app-boundary-marginal}.
Finally, $p(t\text{ is a boundary}\mid y,k)=\sum_{r=1}^{k-1}p(t_r=t\mid y,k)$, giving the result.
\end{proof}

\subsection{Hazard prior boundary cases}
\label{app:hazard-cases}

The length-aware prior family includes hazard-process priors as a useful motivation.
It is sometimes useful to understand the limiting behavior as the hazard vanishes or becomes certain.
The following is an unconditioned prior-only statement; after conditioning on a fixed segment count
and endpoint, some regular-grid hazard factors can become constant over admissible segmentations.

\begin{proposition}[Degenerate hazard limits, prior-only]
\label{prop:hazard-limits}
Consider the index-uniform design $x_t=t$ and a constant hazard $\rho\in(0,1)$ that induces a geometric length prior
$g(\ell)=\rho(1-\rho)^{\ell-1}$ on segment lengths (up to the constraint that the last segment ends at $n$).
These limits are statements about the prior, and therefore about the posterior only insofar as
the data likelihood does not override them. Then:
\begin{enumerate}[label=(\roman*)]
\item As $\rho\downarrow 0$, the prior concentrates on $k=1$ (a single segment) and $p(t\mid k)$ becomes irrelevant.
\item As $\rho\uparrow 1$, the prior concentrates on $k=n$ with unit-length segments, i.e., $t_q=q$ for all $q$.
\end{enumerate}
\end{proposition}

\begin{proof}
For the geometric prior, $\mathbb{P}(L=1)=\rho$ and $\mathbb{P}(L>1)=(1-\rho)$.
As $\rho\downarrow 0$, $\mathbb{P}(L>1)\to 1$ so the probability that a new segment starts at any interior index vanishes,
which forces a single segment spanning $\{1,\dots,n\}$.
As $\rho\uparrow 1$, $\mathbb{P}(L=1)\to 1$ and therefore all segments have length one almost surely, giving $t_q=q$.
\end{proof}

\subsection{Latent groups: identifiability and label switching}
\label{app:label-switching}

In exchangeable latent-group segmentation formulations, the group labels are arbitrary.
When the prior on segmentations and within-group hyperparameters is exchangeable across groups,
posterior inference is invariant under permutations of the group labels. This familiar
\emph{label switching} phenomenon motivates deterministic reporting conventions for the
finite-template EM model of Section~\ref{sec:latent-em}, even though that model optimizes a
finite template-mixture objective rather than sampling a full posterior over all templates.

\begin{proposition}[Permutation invariance]
\label{prop:perm-invariance}
Assume a $G$-group latent model with an exchangeable prior over group-specific parameters and segmentations.
For every permutation $\sigma$ of $\{1,\dots,G\}$,
\[
p\big((t^{(g)},\psi_g)_{g=1}^G\big)
=
p\big((t^{(\sigma(g))},\psi_{\sigma(g)})_{g=1}^G\big).
\]
Then the joint posterior over $(t^{(g)},\psi_g)_{g=1}^G$ and group assignments $z_{1:S}$ satisfies the same invariance.
In particular, any MAP assignment is non-unique up to label permutation.
\end{proposition}

\begin{proof}
The likelihood factors by subjects given assignments and group-specific parameters, and the prior is invariant by assumption.
Therefore the unnormalized posterior is invariant under simultaneous permutation of group labels in both parameters and assignments.
Normalizing preserves invariance.
\end{proof}

\begin{remark}
Permutation invariance does not prevent accurate segmentation or clustering; it only implies that the label names carry no meaning.
For reporting, impose an ordering constraint (e.g., by evidence, segment count, or a canonical parameter summary) or
post-process posterior samples to align labels. For the EM template mixture, record the deterministic
label ordering and deterministic tie-breaking rule with the run metadata.
The EM algorithm in Section~\ref{sec:latent-em} can converge to any one of the symmetric modes.
\end{remark}

\paragraph{Proof of Proposition~\ref{prop:latent-identifiability}.}
Write the marginal subject-level density induced by parameters $(\pi,\tau_1,\dots,\tau_G)$ under
the integrated-block formulation as
\[
m(y; \pi, \tau_{1:G}) = \sum_{g=1}^G \pi_g \, p(y\mid \tau_g),
\qquad
p(y\mid \tau_g) = \frac{p(k_g)}{C_{k_g}} \prod_{q=1}^{k_g} A^{(0)}_{t^{(g)}_{q-1} t^{(g)}_q}(y),
\]
where the block-evidence factors are strictly positive on the declared support by hypothesis (I3).
At the population level, observing the mixture law determines the function
$m(\cdot;\pi,\tau_{1:G})$. Suppose another parameter tuple induces the same population density:
\[
\sum_{g=1}^G \pi_g p(\cdot\mid\tau_g)
=
\sum_{g=1}^G \pi'_g p(\cdot\mid\tau'_g).
\]
Move all terms to one side and combine equal component densities. Hypothesis (I2) gives linear
independence of the component densities in the admissible template family, so every combined
coefficient must be zero. Strictly positive weights then imply that the two representations contain
the same unordered collection of component densities with the same weights. Because the template map
$\tau\mapsto p(\cdot\mid\tau)$ is restricted to the declared identifiable component family, equality
of component densities identifies the corresponding templates. Hence the unordered multiset
$\{(\pi_g,\tau_g)\}_{g=1}^G$ is determined, which is exactly identifiability up to a simultaneous
permutation of group labels. This argument is population-level; finite samples only estimate the
mixture law and do not remove the permutation symmetry. \hfill$\square$

\begin{example}[Label-switching counterexample]
\label{ex:label-switch-counterexample}
The permutation indeterminacy in Proposition~\ref{prop:latent-identifiability} is binding even at
the smallest nontrivial scale. Take $G=2$, $S=2$, a single block family with a symmetric prior, and
two distinct templates $\tau_1=(k_1,t^{(1)})$, $\tau_2=(k_2,t^{(2)})$. Set $\pi_1=\pi_2=1/2$. Then
the two parameter tuples
\[
(\pi,\tau_1,\tau_2) = \big(\tfrac12, \tfrac12; (k_1,t^{(1)}), (k_2,t^{(2)})\big)
\quad\text{and}\quad
(\pi,\tau_1,\tau_2)' = \big(\tfrac12, \tfrac12; (k_2,t^{(2)}), (k_1,t^{(1)})\big)
\]
induce identical marginals over $(y^{(1)},y^{(2)})$. No estimator (Bayes, ML, EM, or otherwise)
that observes only $(y^{(1)},y^{(2)})$ can distinguish them. A deterministic anchoring convention
(e.g.\ ``report templates in nondecreasing $k_g$, breaking ties lexicographically on $t^{(g)}$'') is
therefore required for reproducible label-level reporting.
\end{example}

\subsection{Proof of the max-sum DP correctness (joint MAP segmentation)}
\label{app:max-sum-proof}

We supply here the full correctness proof for the max-sum recursion
$M_{k+1,j}=\max_h\{M_{k,h}+\log\widetilde A^{(0)}_{h,j}\}$ referenced in
Theorem~\ref{thm:dp-correctness}, item 3.

\begin{proposition}[Max-sum correctness]
For every $k\le k_{\max}$ and $j\ge k$, $M_{k,j}$ equals the maximum, over all admissible
$k$-segmentations of $y_{1:j}$, of $\sum_{q=1}^k \log\widetilde A^{(0)}_{t_{q-1},t_q}$.
Consequently, for fixed $k$, $M_{k,n}$ is the optimal block-score sum for that segment count, and backtracking via
$h^\star(k,j)$ recovers one fixed-$k$ joint MAP boundary vector. When selecting across $k$, the
count-prior and normalizer offset $\log p(k)-\log C_k$ must be added to the terminal score.
\end{proposition}

\begin{proof}
Induction on $k$. For $k=1$, $M_{1,j}=\log\widetilde A^{(0)}_{0,j}$, which is the only admissible
$1$-segmentation score. Assume the claim holds for $k$; every admissible $(k+1)$-segmentation of
$y_{1:j}$ decomposes uniquely as a $k$-segmentation of $y_{1:h}$ for some $h\in\{k,\dots,j-1\}$
followed by a terminal block $(h,j]$. By inductive hypothesis the optimal prefix score is
$M_{k,h}$, so the optimal total score over $(k+1)$-segmentations is
$\max_h\{M_{k,h}+\log\widetilde A^{(0)}_{h,j}\}=M_{k+1,j}$. Backtracking recovers the optimum
path by following $h^\star$.
\end{proof}

\subsection{Variational bounds for non-conjugate blocks}
\label{app:variational-bounds}

The routine-level statements in Proposition~\ref{prop:uniform-bounds} are conditional sufficient
bounds rather than automatic guarantees. They require a declared finite reachable block class,
positive exact block evidences on that class, and routine diagnostics that certify the relevant
curvature, tail, ELBO-gap, quadrature-remainder, or EP-residual condition. For quadrature, the
smoothness order $s$ in Proposition~\ref{prop:uniform-bounds} is a regularity parameter distinct from
any moment index. For EP, the residual condition includes positive cavities, retained site
normalizers, and a declared bound on the final moment-matching residual. The resulting
$\varepsilon$ is the supremum of the certified block-local errors over that declared class.

We recall two block-level variational bounds used in Section~\ref{sec:nonconj}.

\paragraph{Jensen's inequality for ELBOs.}
For any density $q(\theta)$ and any block $(i,j]$,
$\log A^{(0)}_{ij}\ge \int q(\theta)\log[p(y_{i+1:j}\mid\theta)\pi(\theta)/q(\theta)]\,d\theta$.
This is the block ELBO. Optimizing over $q$ in a parametric class (e.g., Gaussian) yields a
lower bound on $\log A^{(0)}_{ij}$, hence an upper bound on $-\log A^{(0)}_{ij}$ error.

\paragraph{Jaakkola--Jordan logistic bound.}
For logistic terms, the bound used in Section~\ref{sec:nonconj} is
\[
\log\sigma(z)\ge \log\sigma(\xi)+\frac{z-\xi}{2}-\lambda(\xi)(z^2-\xi^2),
\qquad \lambda(\xi)=\frac{\tanh(\xi/2)}{4\xi}.
\]
Applying this to the signed binomial-logistic linear predictors yields a quadratic surrogate in
$\theta$ that can be combined with a Gaussian prior to produce closed-form coordinate updates.
Iterating on $\{\xi_t\}$ by $\xi_t^\star=\sqrt{\mathbb{E}_q[\theta^2]}$ gives monotone improvement
in the block ELBO.

\subsection{Derivation of P\'olya--Gamma mean-field updates}
\label{app:pg-mf-derivation}

We derive the closed-form updates stated in Proposition~\ref{prop:pg-mf}. Starting from
\eqref{eq:pg-cond-quad}, consider the mean-field ELBO
\[
\mathcal{L}(q_\theta,q_\omega)=\mathbb{E}_q[\log p(y,\theta,\omega)]-\mathbb{E}_q[\log q_\theta(\theta)q_\omega(\omega)].
\]
Coordinate ascent over $q_\theta$ at fixed $q_\omega$ gives $q_\theta^\star(\theta)\propto\exp\mathbb{E}_{q_\omega}[\log p(\theta,\omega,y)]$, which reduces to
$\exp\{-\tfrac12\theta^2(\rho^{-2}+\sum_t \mathbb{E}_q[\omega_t])+\theta(\rho^{-2}\nu+\sum_t \kappa_t)\}$, a Gaussian with the stated $(\mu_q,\sigma_q^2)$.
Similarly, $q_\omega^\star(\omega_t)\propto\exp\mathbb{E}_{q_\theta}[\log p(\omega_t,\theta,y)]\propto p(\omega_t)\exp(-\tfrac{1}{2}\mathbb{E}_q[\theta^2]\omega_t)$, which, by the defining exponential tilting of the PG distribution, equals $\mathrm{PG}(b_t,c_t)$ with $c_t^2=\mathbb{E}_q[\theta^2]=\mu_q^2+\sigma_q^2$. The mean of $\mathrm{PG}(b,c)$ is $\tfrac{b}{2c}\tanh(c/2)$. Monotonicity follows because each coordinate update maximizes its respective factor of $\mathcal{L}$.

\subsection{Complexity proofs}
\label{app:complexity-proofs}

The complexity claims in Theorem~\ref{thm:dp-complexity} were proved by counting operations; we
repeat here the argument for the EM-with-pooled-blocks combination.

\begin{proposition}[EM + pooled-block complexity]
Let $S$ be the number of subjects and $G$ the number of latent groups. One EM iteration of
Algorithm~\ref{alg:multi-em} runs in time
$\mathcal{O}(SGk_{\max}+GSn^2+Gk_{\max}n^2)$ after a one-time precomputation of per-subject
log-block evidences costing $\mathcal{O}(Sn^2)$. If responsibility-weighted score matrices are
formed in a streamed fashion, the $GSn^2$ term can be traded for additional passes over cached
subject matrices. Memory is $\mathcal{O}(Sn^2+Gn^2+Gk_{\max}n)$ under full score storage and can be
reduced by streaming at the cost of recomputation.
\end{proposition}

\begin{proof}
The E-step evaluates $G$ template scores per subject, each requiring at most $k_{\max}$ block
lookups, giving $\mathcal{O}(SGk_{\max})$. The M-step forms $G$ responsibility-weighted score
matrices; with cached subject block-evidence matrices this costs $\mathcal{O}(GSn^2)$ in the
straightforward implementation. Each group then runs a max-sum DP in $\mathcal{O}(k_{\max}n^2)$.
The stated memory bound separates per-subject block storage, group score storage, and DP/backpointer
storage.
\end{proof}
