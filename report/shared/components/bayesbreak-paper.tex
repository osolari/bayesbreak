% tikz-bayesnet emits standalone semicolon tokens in some macros; these
% equivalent definitions draw the same graph without nullfont glyph warnings.
\renewcommand{\edge}[3][]{%
  \foreach \x in {#2}{%
    \foreach \y in {#3}{%
      \draw[->,#1] (\x) -- (\y);%
    }%
  }%
}

}
