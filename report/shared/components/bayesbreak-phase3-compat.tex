% tikz-bayesnet 0.1 terminates \foreach blocks with standalone semicolon
% tokens, which pdfTeX reports as missing characters in nullfont. These
% equivalent local definitions retain the drawing paths but remove those
% non-drawing tokens so clean builds have no false missing-glyph warnings.
\renewcommand{\edge}[3][]{%
  \foreach \x in {#2}{%
    \foreach \y in {#3}{%
      \draw[->,#1] (\x) -- (\y);%
    }%
  }%
}
\renewcommand{\plate}[4][]{%
  \node[wrap=#3] (#2-wrap) {};%
  \node[plate caption=#2-wrap] (#2-caption) {#4};%
  \node[plate=(#2-wrap)(#2-caption),#1] (#2) {};%
}
