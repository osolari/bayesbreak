% Shared TikZ vocabulary for paper, slides, and book.
% Requires: colors.tex loaded before this file.
% Requires: tikz loaded with libraries: positioning, arrows.meta, calc.
%
% Design rules:
%   - No rounded corners on conceptual nodes.
%   - No visible node outlines (draw=none).
%   - Solid fills with semantic colors from colors.tex.
%   - Relative node positioning throughout.
%   - Clean thin arrows with consistent arrowheads.
\usetikzlibrary{positioning,arrows.meta,calc}

\tikzset{
  % ------------------------------------------------------------------
  % BayesBreak primary node styles (flat, borderless, square-cornered)
  % ------------------------------------------------------------------
  bbbase/.style={
    rectangle,
    draw=none,
    rounded corners=0pt,
    minimum height=9mm,
    align=center,
    inner sep=5pt,
    font=\small\sffamily,
  },
  % Primary stage node — dark fill, white text
  bbstage/.style={
    bbbase,
    fill=bbStageFill,
    text=bbStageText,
    minimum width=28mm,
  },
  % Secondary node — light fill, dark text
  bbnode/.style={
    bbbase,
    fill=bbNodeFill,
    text=bbNodeText,
    minimum width=28mm,
  },
  % Accent / result node — rose fill, white text
  bbaccent/.style={
    bbbase,
    fill=bbAccentFill,
    text=white,
    minimum width=28mm,
  },
  % Neutral background node — light grey
  bbneutral/.style={
    bbbase,
    fill=bbLightGrey,
    text=bbNodeText,
    minimum width=28mm,
  },
  % ------------------------------------------------------------------
  % Legacy styles kept for backward compatibility with existing diagrams.
  % Gradually migrate to bbstage / bbnode / bbaccent.
  % ------------------------------------------------------------------
  boxbase/.style={
    rectangle,
    draw=none,
    rounded corners=0pt,
    minimum height=9mm,
    align=center,
    inner sep=5pt,
    font=\small\sffamily,
  },
  primarybox/.style={boxbase, fill=bbStageFill, text=white, minimum width=28mm},
  secondarybox/.style={boxbase, fill=bbNodeFill, text=bbNodeText, minimum width=28mm},
  neutralbox/.style={boxbase, fill=bbLightGrey, text=bbNodeText, minimum width=28mm},
  successbox/.style={boxbase, fill=bbAccentFill, text=white, minimum width=28mm},
  % ------------------------------------------------------------------
  % Arrow styles
  % ------------------------------------------------------------------
  bbarrow/.style={
    -{Stealth[length=2.2mm]},
    line width=0.75pt,
    draw=bbBlack,
  },
  bbarrowdash/.style={
    -{Stealth[length=2.2mm]},
    line width=0.75pt,
    draw=bbBlack,
    dashed,
  },
  % Legacy arrow aliases
  arr/.style={bbarrow},
  arrPrimary/.style={bbarrow},
  arrSecondary/.style={bbarrow},
  arrow/.style={bbarrow},
  darrow/.style={bbarrowdash},
  % ------------------------------------------------------------------
  % Other reusable styles
  % ------------------------------------------------------------------
  tier/.style={draw=bbGrey, dashed, line width=0.5pt, inner sep=3mm},
  % Legacy cs* styles (used in older diagrams — kept for compatibility)
  csblock/.style={bbnode, minimum height=1.0cm, minimum width=3.6cm},
  csfinal/.style={bbaccent, minimum height=1.0cm, minimum width=3.6cm},
  cslevel1/.style={bbstage, text width=9em, minimum height=2.2em, font=\small\rmfamily\bfseries},
  cslevel2/.style={bbnode, text width=7.5em, minimum height=1.8em, font=\scriptsize\rmfamily},
  cscrosscut/.style={bbneutral, text width=8em, minimum height=1.8em, font=\scriptsize\rmfamily},
  csmethodbox/.style={bbstage, minimum width=2.6cm, minimum height=0.8cm, font=\small\rmfamily\bfseries},
  cspropbox/.style={bbnode, minimum width=2.6cm, minimum height=0.6cm, font=\scriptsize\rmfamily, text width=2.6cm},
  csstagelbl/.style={font=\rmfamily\bfseries\small, text=bbRose},
}
